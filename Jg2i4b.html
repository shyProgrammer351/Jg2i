문제

함수 호출시 값을 전달받고, 함수 내에서 기본자료형 수정, 객체 값을 직접 수정, 새 객체를 생성하여 수정하는 프로그램입니다 이를 해결하는 다음 프로그램에 대해 빈칸을 채우세요 
uppercase() 호출전   argument1 = a     argument2 = d     argument3 = g 
uppercase() 변경전  parameter1 = a    parameter2 = d    parameter3 = g 
uppercase() 변경후  parameter1 = A    parameter2 = D    parameter3 = G 
uppercase() 호출후   argument1 = a     argument2 = d     argument3 = G

코드 빈칸 채우기

 Char  
{  
	 char value;  
	  
	 Char( char value ) {  
		.value = value;  
	}  
	  
	 String () {  
		 .value + " ";  
	}  
}  
	  
  PassByValueVsPassByReference  
{  
	 static void uppercase( char parameter1, Char parameter2, Char parameter3 ) {  
		System.out.println( "uppercase() 변경전  parameter1 = " + parameter1 + "    parameter2 = " + parameter2 + "   parameter3 = " + parameter3 );  
	  
		parameter1 = 'A';  
		parameter2 =  
		              Char( 'D' );  
		parameter3.value = 'G';  
	  
		System.out.println( "uppercase() 변경후  parameter1 = " + parameter1 + "    parameter2 = " + parameter2 + "   parameter3 = " + parameter3 );  
	}  
	  
	public static void main( String[] args ) { 
		char argument1 = 'a';  
		Char argument2 =  
		                 Char( 'd' );  
		Char argument3 =  
		                 Char( 'g' );  
	  
		System.out.println( "uppercase() 호출전   argument1 = " +  argument1 + "     argument2 = " +  argument2 + "    argument3 = " +  argument3 );  
		uppercase( argument1, argument2, argument3 );  
		System.out.println( "uppercase() 호출후   argument1 = " +  argument1 + "     argument2 = " +  argument2 + "    argument3 = " +  argument3 );  
	  
	}
